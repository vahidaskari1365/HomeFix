   180→// Stats strip — GET /api/stats
   181→// ============================================================
   182→function StatsStrip() {
   183→  const { data, isLoading, isError, refetch } = useQuery({
   184→    queryKey: ["stats"],
   185→    queryFn: () => apiGet<{ stats: StatsDTO }>("/api/stats"),
   186→    retry: 1,
   187→  });
   188→
   189→  const stats = data?.stats;
   190→  const items = [
   191→    { label: "متخصص فعال", value: stats ? toFa(stats.specialistsCount) : "—" },
   192→    { label: "سفارش موفق", value: stats ? toFa(stats.ordersCount) : "—" },
   193→    { label: "میانگین امتیاز", value: stats ? `${toFa(stats.avgRating.toFixed(1))} ★` : "—" },
   194→    { label: "دسته خدمات", value: toFa(8) },
   195→  ];
   196→
   197→  return (
   198→    <section aria-label="آمار پلتفرم" className={`${CONTAINER} py-6 md:py-8`}>
   199→      {isError ? (
   200→        <ErrorState message="خطا در دریافت آمار" onRetry={() => refetch()} />
   201→      ) : (
   202→        <div className="grid grid-cols-2 gap-3 md:grid-cols-4 md:gap-4">
   203→          {items.map((item) => (
   204→            <div key={item.label} className={`${CARD} p-5 text-center`}>
   205→              {isLoading ? (
   206→                <div className="mx-auto mb-2 h-7 w-14 animate-pulse rounded bg-muted" />
   207→              ) : (
   208→                <p className="text-2xl font-extrabold text-primary">{item.value}</p>
   209→              )}
   210→              <p className="mt-1 text-xs font-medium text-muted-foreground md:text-sm">{item.label}</p>
   211→            </div>
   212→          ))}
   213→        </div>
   214→      )}
   215→    </section>
   216→  );
   217→}

   220→// AI assistant — POST /api/ai/suggest
   221→// ============================================================
   222→function AiAssistant() {
   223→  const openBooking = useHomeFix((s) => s.openBooking);
   224→  const [text, setText] = useState("");
   225→
   226→  const suggest = useMutation({
   227→    mutationFn: (t: string) =>
   228→      apiPost<{ suggestion: AiSuggestionDTO | null }>("/api/ai/suggest", { text: t }),
   229→  });
   230→
   231→  const suggestion = suggest.data?.suggestion ?? null;
   232→
   233→  return (
   234→    <section aria-label="دستیار هوشمند" className={`${CONTAINER} py-8`}>
   235→      <div className={`${CARD} mx-auto max-w-3xl p-6 md:p-8`}>
   236→        <div className="mb-4 flex items-center gap-3">
   237→          <span className="grid size-11 place-items-center rounded-xl bg-accent/15 text-accent-foreground">
   238→            <Sparkles className="size-5" />
   239→          </span>
   240→          <div>
   241→            <h2 className="font-extrabold">نمی‌دانی چه سرویسی لازم داری؟</h2>
   242→            <p className="text-xs text-muted-foreground">مشکل را بنویس، دستیار هوشمند پیشنهاد می‌دهد</p>
   243→          </div>
   244→        </div>
   245→
   246→        <Textarea
   247→          value={text}
   248→          onChange={(e) => setText(e.target.value)}
   249→          placeholder="مثلاً: کولرم روشن می‌شه ولی خنک نمی‌کنه…"
   250→          rows={3}
   251→          className="resize-none"
   252→          aria-label="توصیف مشکل"
   253→        />
   254→
   255→        <div className="mt-4 flex flex-wrap items-center gap-3">
   256→          <Button
   257→            className="min-h-11 bg-accent text-accent-foreground hover:bg-accent/90"
   258→            disabled={suggest.isPending || text.trim().length < 3}
   259→            onClick={() => suggest.mutate(text.trim())}
   260→          >
   261→            <Sparkles className="size-4" />
   262→            {suggest.isPending ? "در حال فکر کردن…" : "دستیار هوشمند HomeFix"}
   263→          </Button>
   264→          {suggest.isError ? (
   265→            <p className="text-sm text-muted-foreground">
   266→              الان نمی‌تونم پیشنهاد بدم؛ از دسته‌بندی‌ها انتخاب کن
   267→            </p>
   268→          ) : null}
   269→        </div>
   270→
   271→        {suggestion ? (
   272→          <motion.div
   273→            initial={{ opacity: 0, y: 10 }}
   274→            animate={{ opacity: 1, y: 0 }}
   275→            className="mt-5 rounded-xl border border-accent/50 bg-accent/10 p-4"
   276→          >
   277→            <p className="text-sm leading-relaxed">
   278→              <span className="font-extrabold">پیشنهاد ما: {suggestion.serviceName}</span>
   279→              <span className="text-muted-foreground"> — {suggestion.reason}</span>
   280→            </p>
   281→            <Button
   282→              className="mt-3 min-h-11"
   283→              onClick={() => openBooking({ serviceId: suggestion.serviceId, aiText: text.trim() })}
   284→            >
   285→              شروع درخواست با این سرویس
   286→              <ArrowLeft className="size-4" />
   287→            </Button>
   288→          </motion.div>
   289→        ) : null}
   290→      </div>
   291→    </section>
   292→  );
   293→}
